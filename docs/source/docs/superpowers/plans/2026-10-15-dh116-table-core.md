import pytest
from go2_pvcnn.control.dh116.table_core import AngleTable

def test_values_and_derivative_share_interpolant():

    y = x*x + x

- [ ] 最小实现如下；不增加任何坐标转换默认值。

```python
import numpy as np
from scipy.interpolate import PchipInterpolator

class AngleTable:
    def __init__(self, x_rad, y_rad):
        x = np.array(x_rad, dtype=np.float64, copy=True)
        y = np.array(y_rad, dtype=np.float64, copy=True)
        if (x.ndim != 1 or y.shape != x.shape or x.size < 2
                or not np.isfinite(x).all() or not np.isfinite(y).all()
                or np.any(np.diff(x) <= 0)):
            raise ValueError('expected finite strictly increasing paired samples')
        self._domain = (float(x[0]), float(x[-1]))
        self._value = PchipInterpolator(x, y, extrapolate=False)
        self._derivative = self._value.derivative()

    def evaluate(self, q_rad):
        q = np.asarray(q_rad, dtype=np.float64)
        lo, hi = self._domain
        if (not np.isfinite(q).all() or np.any(q < lo) or np.any(q > hi)):
            raise ValueError('query outside measured table domain')
        value, derivative = self._value(q), self._derivative(q)
        if not np.isfinite(value).all() or not np.isfinite(derivative).all():
            raise ValueError('non-finite interpolation result')
        return value, derivative
```

```python
import hashlib
from pathlib import Path
import xml.etree.ElementTree as ET
import zipfile
import numpy as np
from .table_core import AngleTable

NS = {'m': 'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}

def read_samples(path, expected_sha256, expected_headers, expected_count):
    raw = Path(path).read_bytes()
    if hashlib.sha256(raw).hexdigest() != expected_sha256:
        raise ValueError('source SHA mismatch')
    import io
    with zipfile.ZipFile(io.BytesIO(raw)) as z:
        strings = []
        if 'xl/sharedStrings.xml' in z.namelist():
            root = ET.fromstring(z.read('xl/sharedStrings.xml'))
            strings = [''.join(t.text or '' for t in n.findall('.//m:t', NS))
                       for n in root.findall('m:si', NS)]
        root = ET.fromstring(z.read('xl/worksheets/sheet1.xml'))
        rows = root.findall('.//m:sheetData/m:row', NS)
        values, header_seen = [], False
        for row in rows:
            r = int(row.attrib['r'])
            cells = {c.attrib['r']: c for c in row.findall('m:c', NS)}
            if r < 4:
                continue
            pair = []
            for col in ('B', 'C'):
                c = cells.get(f'{col}{r}')
                if c is None:
                    raise ValueError('missing B/C cell')
                v = c.find('m:v', NS)
                if v is None or v.text is None:
                    raise ValueError('missing cell value')
                if r == 4:
                    pair.append(strings[int(v.text)] if c.get('t') == 's' else v.text)
                else:
                    if c.get('t', 'n') != 'n' or c.find('m:f', NS) is not None:
                        raise ValueError('expected literal numeric samples')
                    pair.append(float(v.text))
            if r == 4:
                if header_seen or tuple(pair) != tuple(expected_headers):
                    raise ValueError('table header mismatch')
                header_seen = True
            else:
                values.append(pair)
        if not header_seen or len(values) != expected_count:
            raise ValueError('sample count/header mismatch')
    samples = np.deg2rad(np.asarray(values, dtype=np.float64))
    AngleTable(samples[:, 0], samples[:, 1])
    return samples[:, 0].copy(), samples[:, 1].copy()
```
